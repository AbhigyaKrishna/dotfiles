# --- power: backs meta/system/tlp.d/10-local.conf --------------------------
# TLP is the power manager, not power-profiles-daemon. On this laptop there is
# no ACPI platform_profile, so PPD had only the amd_pstate knob and reported
# `PlatformDriver: placeholder` for two of its three profiles; TLP also tunes
# disk APM, PCIe ASPM, wifi power save and USB. TLP switches on AC/battery
# natively, which is why no udev rule is needed for it.
tlp
tlp-pd                      # provides= power-profiles-daemon and conflicts
                            #   with it. Ships the UPower.PowerProfiles and
                            #   net.hadess.PowerProfiles D-Bus interfaces, so
                            #   noctalia's widget (Quickshell.Services.UPower)
                            #   keeps working. CLI is tlpctl, NOT
                            #   powerprofilesctl — that binary is gone.
tlpui                       # GTK editor for the drop-in, the way xdgctl edits
                            #   mimeapps.list — changes show up as a diff

# TLP tunes the OS-visible knobs; it cannot touch the SMU. This chassis needs
# that second layer: the firmware targets Tjmax (95°C) and will sit on it, which
# means throttling reactively — overshoot, cut hard, recover, overshoot again,
# with frametimes following the sawtooth. ryzenadj lowers the junction
# temperature target and the power budget directly so the part settles below the
# cliff instead. This is the backstop, not the fix — nbfc-linux below is what
# actually made this machine cool. Backs meta/system/ryzenadj-thermal.service —
# AUR, so paru rather than pacman, and not covered by install-packages.sh.
ryzenadj                    # AUR. Writes are volatile and lost on resume,
                            #   which is why the unit re-applies on wake

# Fan control, because the stock curve on this chassis does not ramp. Measured:
# Tctl 74.8°C at idle and 85–96°C under load, with the fans audibly turning but
# never spinning up. hp-wmi is no help — its fan1_input/fan2_input read a flat 0
# in every state, and forcing pwm1_enable=0 (its "boost" mode) moved a pinned
# all-core load by 0.8°C, which is noise. The driver's EC offsets are wrong for
# this model, so both the readings and the control are inert.
#
# nbfc-linux talks to the EC directly instead, and ships a config matched to this
# exact machine — "HP Pavilion Gaming Laptop 15-ec2xxx", which is verbatim what
# /sys/class/dmi/id/product_name reports. It drives the CPU and GPU fans
# separately (EC registers 45 and 44) and its curve asks for 80% at 75°C and
# 100% above, where the stock behaviour asks for almost nothing.
nbfc-linux                  # AUR. Unit is nbfc_service.service, config lands
                            #   at /etc/nbfc/nbfc.json via `nbfc config -s`

# --- maintenance ------------------------------------------------------------
# Back the timers in meta/systemd-system-units.txt. pacman-contrib provides
# paccache.timer, btrfs-progs the scrub timer, snapper and snap-pac the
# snapshots that limine-snapper-sync turns into boot entries. All were already
# installed here; they are listed so a fresh machine gets the timers too.
pacman-contrib
btrfs-progs
snapper
snap-pac
limine-snapper-sync
fwupd                       # firmware updates: fwupdmgr refresh, then
                            #   fwupdmgr update. Nothing else was delivering
                            #   them on this machine. fwupd-refresh.timer in
                            #   meta/systemd-system-units.txt keeps the LVFS
                            #   metadata current.
                            #
                            #   Two things to know before running `update`.
                            #   LVFS has nothing for this laptop's BIOS — HP
                            #   does not publish the Pavilion Gaming 15-ec2xxx
                            #   there — so System Firmware reporting no updates
                            #   means uncovered, not current.
                            #
                            #   And the UEFI dbx update it does offer is
                            #   deliberately not applied. dbx is the Secure
                            #   Boot revocation list, which firmware only
                            #   consults when Secure Boot is on; it is off here
                            #   (SecureBoot efivar 0, no shim, single OS). The
                            #   update is therefore inert, while being awkward
                            #   to undo — clearing dbx means resetting Secure
                            #   Boot keys in BIOS setup — and dbx is the usual
                            #   culprit when fwupd breaks a boot. If Secure
                            #   Boot is ever enabled, apply dbx first and set
                            #   up signing (sbctl) after, not the reverse.
